# Funcion auxiliar base R para Frequency Diagram y Stereogram.
# Copiala una vez; luego solo cambias los datos de las plantillas de abajo.
draw_prisms_3d <- function(x_left,x_right,y_left,y_right,height,
                           xlab="X",ylab="Y",zlab="Frequency",
                           main="3D Diagram") {
  zmax <- max(height,na.rm=TRUE)
  if (!is.finite(zmax) || zmax<=0) zmax <- 1

  P <- persp(
    x=c(min(x_left),max(x_right)),
    y=c(min(y_left),max(y_right)),
    z=matrix(0,2,2),
    xlim=c(min(x_left),max(x_right)),
    ylim=c(min(y_left),max(y_right)),
    zlim=c(0,zmax*1.15),
    theta=40, phi=25, expand=0.8,
    xlab=xlab, ylab=ylab, zlab=zlab, main=main
  )

  cells <- expand.grid(i=seq_along(x_left),j=seq_along(y_left))
  cells$order <- x_left[cells$i] + y_left[cells$j]
  cells <- cells[order(cells$order,decreasing=TRUE),]

  face <- function(xs,ys,zs) {
    p <- trans3d(xs,ys,zs,P)
    polygon(p$x,p$y,border="black")
  }

  for (k in seq_len(nrow(cells))) {
    i <- cells$i[k]; j <- cells$j[k]
    h <- height[i,j]
    if (!is.finite(h) || h<=0) next

    xl <- x_left[i]; xr <- x_right[i]
    yl <- y_left[j]; yr <- y_right[j]

    face(c(xl,xr,xr,xl),c(yl,yl,yl,yl),c(0,0,h,h))
    face(c(xr,xr,xr,xr),c(yl,yr,yr,yl),c(0,0,h,h))
    face(c(xl,xl,xl,xl),c(yl,yr,yr,yl),c(0,0,h,h))
    face(c(xl,xr,xr,xl),c(yr,yr,yr,yr),c(0,0,h,h))
    face(c(xl,xr,xr,xl),c(yl,yl,yr,yr),c(h,h,h,h))
  }
}
